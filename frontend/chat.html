How to get a Privy token (React) 
// 1) Create app at https://dashboard.privy.io
// 2) Same PRIVY_APP_ID as kit backend (src/.env.local)
// 3) Enable Solana embedded / linked wallet for your users

import { PrivyProvider, usePrivy } from "@privy-io/react-auth";

<PrivyProvider appId={import.meta.env.VITE_PRIVY_APP_ID}>
  <App />
</PrivyProvider>

// In a component:
const { ready, authenticated, login, getAccessToken } = usePrivy();
await login();
const token = await getAccessToken();
// paste token above, or set Authorization: Bearer ${token} on /stream
